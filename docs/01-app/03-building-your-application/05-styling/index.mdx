"use client";

import { useState, useEffect } from "react";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Globe, Mail, PackageSearch, Phone } from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";
import Image from "next/image";

const MotionDiv = motion.div;

export default function Home() {
  const [trackingCode, setTrackingCode] = useState("");
  const [language, setLanguage] = useState("it");
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const timeout = setTimeout(() => setLoading(false), 1500);
    return () => clearTimeout(timeout);
  }, []);

  const handleTracking = () => {
    alert(`${language === "it" ? "Tracking per" : "Tracking for"}: ${trackingCode}`);
  };

  const t = {
    it: {
      title: "Spedizioni Rusciano",
      headline: "Servizi di spedizione affidabili e veloci",
      subtitle: "Traccia le tue spedizioni, richiedi un preventivo e contattaci per ogni necessità.",
      trackTitle: "Traccia la tua spedizione",
      trackPlaceholder: "Inserisci il codice di tracciamento",
      trackButton: "Traccia",
      quoteTitle: "Richiedi un preventivo",
      from: "Città di partenza",
      to: "Città di destinazione",
      weight: "Peso del pacco (kg)",
      dimensions: "Dimensioni (cm)",
      quoteButton: "Calcola Preventivo",
      contactTitle: "Contattaci",
      contactText: "Hai bisogno di assistenza? Contattaci tramite email o telefono."
    },
    en: {
      title: "Rusciano Shipping",
      headline: "Reliable and fast shipping services",
      subtitle: "Track your shipments, request a quote, and contact us for assistance.",
      trackTitle: "Track your shipment",
      trackPlaceholder: "Enter tracking code",
      trackButton: "Track",
      quoteTitle: "Request a quote",
      from: "Departure city",
      to: "Destination city",
      weight: "Package weight (kg)",
      dimensions: "Dimensions (cm)",
      quoteButton: "Get Quote",
      contactTitle: "Contact us",
      contactText: "Need help? Reach us by email or phone."
    }
  };

  const lang = t[language];

  return (
    <main className="min-h-screen bg-gray-100 text-gray-900">
      <AnimatePresence>
        {loading ? (
          <motion.div
            key="loader"
            className="fixed inset-0 z-50 flex items-center justify-center bg-white"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
          >
            <motion.div
              className="text-3xl font-semibold text-gray-800"
              initial={{ opacity: 0, scale: 0.8 }}
              animate={{ opacity: 1, scale: [1, 1.05, 1] }}
              transition={{ duration: 1.2, repeat: Infinity }}
            >
              <Image src="/logo-rusciano.webp" alt="Logo Spedizioni Rusciano" width={220} height={50} />
            </motion.div>
          </motion.div>
        ) : null}
      </AnimatePresence>

      {!loading && (
        <>
          <header className="bg-white shadow p-4 flex justify-between items-center">
            <div className="flex items-center gap-4">
              <Image src="/logo-rusciano.webp" alt="Logo Spedizioni Rusciano" width={160} height={40} />
              <div className="text-2xl font-bold">{lang.title}</div>
            </div>
            <div className="space-x-2">
              <Button variant="ghost" onClick={() => setLanguage("it")}>IT</Button>
              <Button variant="ghost" onClick={() => setLanguage("en")}>EN</Button>
            </div>
          </header>

          <section className="p-6 max-w-4xl mx-auto">
            <MotionDiv initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.8 }}>
              <h1 className="text-3xl font-semibold mb-4">{lang.headline}</h1>
              <p className="mb-6">{lang.subtitle}</p>

              <Card className="mb-6">
                <CardContent className="p-4 space-y-4">
                  <h2 className="text-xl font-medium flex items-center gap-2"><PackageSearch /> {lang.trackTitle}</h2>
                  <div className="flex gap-2">
                    <Input placeholder={lang.trackPlaceholder} value={trackingCode} onChange={(e) => setTrackingCode(e.target.value)} />
                    <Button onClick={handleTracking}>{lang.trackButton}</Button>
                  </div>
                </CardContent>
              </Card>

              <Card className="mb-6">
                <CardContent className="p-4 space-y-4">
                  <h2 className="text-xl font-medium flex items-center gap-2"><Globe /> {lang.quoteTitle}</h2>
                  <form className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <Input placeholder={lang.from} />
                    <Input placeholder={lang.to} />
                    <Input placeholder={lang.weight} />
                    <Input placeholder={lang.dimensions} />
                    <Button className="md:col-span-2">{lang.quoteButton}</Button>
                  </form>
                </CardContent>
              </Card>

              <Card>
                <CardContent className="p-4 space-y-4">
                  <h2 className="text-xl font-medium flex items-center gap-2"><Mail /> {lang.contactTitle}</h2>
                  <p>{lang.contactText}</p>
                  <div className="flex flex-col gap-1">
                    <div className="flex items-center gap-2"><Mail className="w-4" /> info@spedizionirusciano.it</div>
                    <div className="flex items-center gap-2"><Phone className="w-4" /> +39 081 123 4567</div>
                  </div>
                </CardContent>
              </Card>
            </MotionDiv>
          </section>
        </>
      )}
    </main>
  );
}
